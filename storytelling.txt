Slide 31 — What VSA Is Not

Vertical Slice Architecture is not:

❌ Every feature must have the same structure

❌ Every feature must have exactly one handler

❌ No shared code

❌ No domain layer

❌ No abstractions

❌ "Put everything in one folder"

Instead:

Give each feature the structure it actually needs.

Slide 32 — When Does It Make Sense?

VSA can be particularly useful when:

The application has many business features
Features evolve independently
Different features have different complexity
Teams work on different areas
You want changes to remain localized

It also introduces trade-offs:

More files/folders
Potential duplication
Requires discipline around shared code
Can be unnecessary for very small applications

using QuickBite.Api.Entities;
using QuickBite.Api.Repositories;

namespace QuickBite.Api.Services;

public class OrderService : IOrderService
{
    private readonly IOrderRepository _repository;

    public OrderService(IOrderRepository repository)
    {
        _repository = repository;
    }

    public async Task<PlaceOrderResponse> PlaceOrder(
        PlaceOrderRequest request)
    {
        var order = new Order
        {
            Id = Guid.NewGuid(),
            CustomerId = request.CustomerId,
            RestaurantId = request.RestaurantId,
            CreatedAt = DateTime.UtcNow
        };

        foreach (var item in request.Items)
        {
            order.Items.Add(new OrderItem
            {
                Id = Guid.NewGuid(),
                OrderId = order.Id,
                MenuItemId = item.MenuItemId,
                Quantity = item.Quantity,
                UnitPrice = item.UnitPrice
            });
        }

        order.TotalAmount = order.Items.Sum(
            x => x.Quantity * x.UnitPrice);

        await _repository.AddAsync(order);

        return new PlaceOrderResponse(order.Id);
    }
}

Now the handler becomes:

public class PlaceOrderHandler
{
    private readonly QuickBiteDbContext _db;
    private readonly PlaceOrderValidator _validator;
    private readonly OrderPricing _pricing;
    private readonly PaymentProcessor _payment;
    private readonly OrderNotification _notification;

    public PlaceOrderHandler(
        QuickBiteDbContext db,
        PlaceOrderValidator validator,
        OrderPricing pricing,
        PaymentProcessor payment,
        OrderNotification notification)
    {
        _db = db;
        _validator = validator;
        _pricing = pricing;
        _payment = payment;
        _notification = notification;
    }

    public async Task<PlaceOrderResponse> Handle(
        PlaceOrderRequest request)
    {
        _validator.Validate(request);

        var restaurant =
            await GetRestaurant(request.RestaurantId);

        ValidateRestaurant(restaurant);

        var menuItems =
            await GetMenuItems(request);

        ValidateItems(request, menuItems);

        var subtotal =
            CalculateSubtotal(request, menuItems);

        var discount = 0m;

        var price =
            _pricing.Calculate(
                subtotal,
                discount,
                30m);

        await _payment.Process(price.Total);

        var order =
            CreateOrder(
                request,
                menuItems,
                price);

        _db.Orders.Add(order);

        await _db.SaveChangesAsync();

        await _notification.Send(order.Id);

        return new PlaceOrderResponse(order.Id);
    }
}

Domain/Order.cs
namespace QuickBite.Api.Domain;

public class Order
{
    public Guid Id { get; private set; }

    public Guid CustomerId { get; private set; }

    public Guid RestaurantId { get; private set; }

    public decimal TotalAmount { get; private set; }

    public OrderStatus Status { get; private set; }

    private Order()
    {
    }

    public Order(
        Guid customerId,
        Guid restaurantId,
        decimal totalAmount)
    {
        Id = Guid.NewGuid();
        CustomerId = customerId;
        RestaurantId = restaurantId;
        TotalAmount = totalAmount;
        Status = OrderStatus.Pending;
    }

    public void Confirm()
    {
        if (Status != OrderStatus.Pending)
            throw new InvalidOperationException(
                "Only pending orders can be confirmed.");

        Status = OrderStatus.Confirmed;
    }

    public void Cancel()
    {
        if (Status == OrderStatus.Delivered)
            throw new InvalidOperationException(
                "Delivered orders cannot be cancelled.");

        if (Status == OrderStatus.Cancelled)
            throw new InvalidOperationException(
                "Order is already cancelled.");

        Status = OrderStatus.Cancelled;
    }
}

Now the feature says:

order.Cancel();

Instead of knowing the business rule itself.
